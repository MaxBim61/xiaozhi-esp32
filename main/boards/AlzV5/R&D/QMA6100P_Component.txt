В проекте Xiaozhi‑ESP32 плата AlzV5 использует акселерометр QMA6100P, подключённый по I2C с пинами:

SDA → GPIO40

SCL → GPIO41

Задача: настроить детектор движения (any‑motion) так, чтобы при встряхивании или перемещении устройства генерировалось прерывание, которое выведет ESP32‑S3 из глубокого сна.

qma6100p.h – объявление класса/функций драйвера
cpp
#ifndef QMA6100P_H
#define QMA6100P_H

#include "driver/i2c.h"

// I2C адрес (по умолчанию 0x12, если ADDR = GND)
#define QMA6100P_I2C_ADDR        0x12

// Регистры QMA6100P (выдержка из даташита)
#define QMA6100P_REG_CHIP_ID     0x00   // Chip ID = 0xE5
#define QMA6100P_REG_POWER_MODE  0x11   // Mode register
#define QMA6100P_REG_RANGE       0x0F   // Range select
#define QMA6100P_REG_BW          0x10   // Bandwidth / ODR
#define QMA6100P_REG_INT_EN_1    0x16   // Interrupt enable 1
#define QMA6100P_REG_INT_MAP_1   0x19   // Interrupt mapping 1
#define QMA6100P_REG_INT_MAP_2   0x1A   // Interrupt mapping 2
#define QMA6100P_REG_MOTION_CFG  0x09   // Motion configuration
#define QMA6100P_REG_MOTION_TH   0x0A   // Motion threshold
#define QMA6100P_REG_INT_STATUS_0 0x0B  // Interrupt status
#define QMA6100P_REG_ACCEL_X_L    0x01
#define QMA6100P_REG_ACCEL_X_H    0x02
#define QMA6100P_REG_ACCEL_Y_L    0x03
#define QMA6100P_REG_ACCEL_Y_H    0x04
#define QMA6100P_REG_ACCEL_Z_L    0x05
#define QMA6100P_REG_ACCEL_Z_H    0x06

// Значения для настройки
#define QMA6100P_MODE_NORMAL      0x80   // Normal mode (bit7=1)
#define QMA6100P_RANGE_2G         0x01   // ±2g
#define QMA6100P_RANGE_4G         0x02
#define QMA6100P_RANGE_8G         0x03
#define QMA6100P_RANGE_16G        0x04
#define QMA6100P_BW_250HZ         0x09   // 250 Hz
#define QMA6100P_INT_ANY_MOTION_EN (1 << 2) // бит 2 в INT_EN_1
#define QMA6100P_INT1_ANY_MOTION   (1 << 2) // бит 2 в INT_MAP_1

class QMA6100P {
public:
    QMA6100P(i2c_port_t i2c_port, gpio_num_t int1_pin, uint8_t addr = QMA6100P_I2C_ADDR);
    esp_err_t init();
    esp_err_t configureAnyMotion(uint8_t threshold = 20, uint8_t duration = 5);
    esp_err_t enableInterrupt(bool enable);
    esp_err_t readAccel(float &x, float &y, float &z);
    bool isMotionTriggered();

private:
    i2c_port_t _port;
    gpio_num_t _int1_pin;
    uint8_t _addr;
    uint8_t _range;   // текущий диапазон для пересчёта сырых значений

    esp_err_t writeReg(uint8_t reg, uint8_t value);
    esp_err_t readReg(uint8_t reg, uint8_t *value);
    esp_err_t readRegs(uint8_t reg, uint8_t *buf, size_t len);
};

#endif
qma6100p.cpp – реализация
cpp
#include "qma6100p.h"
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"

static const char *TAG = "QMA6100P";

QMA6100P::QMA6100P(i2c_port_t port, gpio_num_t int1_pin, uint8_t addr)
    : _port(port), _int1_pin(int1_pin), _addr(addr), _range(QMA6100P_RANGE_2G) {}

esp_err_t QMA6100P::writeReg(uint8_t reg, uint8_t value) {
    i2c_cmd_handle_t cmd = i2c_cmd_link_create();
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (_addr << 1) | I2C_MASTER_WRITE, true);
    i2c_master_write_byte(cmd, reg, true);
    i2c_master_write_byte(cmd, value, true);
    i2c_master_stop(cmd);
    esp_err_t ret = i2c_master_cmd_begin(_port, cmd, pdMS_TO_TICKS(100));
    i2c_cmd_link_delete(cmd);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "I2C write error: %s", esp_err_to_name(ret));
    }
    return ret;
}

esp_err_t QMA6100P::readReg(uint8_t reg, uint8_t *value) {
    return readRegs(reg, value, 1);
}

esp_err_t QMA6100P::readRegs(uint8_t reg, uint8_t *buf, size_t len) {
    i2c_cmd_handle_t cmd = i2c_cmd_link_create();
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (_addr << 1) | I2C_MASTER_WRITE, true);
    i2c_master_write_byte(cmd, reg, true);
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (_addr << 1) | I2C_MASTER_READ, true);
    i2c_master_read(cmd, buf, len, I2C_MASTER_LAST_NACK);
    i2c_master_stop(cmd);
    esp_err_t ret = i2c_master_cmd_begin(_port, cmd, pdMS_TO_TICKS(100));
    i2c_cmd_link_delete(cmd);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "I2C read error: %s", esp_err_to_name(ret));
    }
    return ret;
}

esp_err_t QMA6100P::init() {
    uint8_t chip_id = 0;
    if (readReg(QMA6100P_REG_CHIP_ID, &chip_id) != ESP_OK) {
        return ESP_FAIL;
    }
    ESP_LOGI(TAG, "Chip ID: 0x%02X", chip_id);
    if (chip_id != 0xE5) { // ожидаемое значение
        ESP_LOGE(TAG, "Invalid chip ID");
        return ESP_ERR_NOT_FOUND;
    }

    // Выход из sleep (если был) – запись в регистр режима
    writeReg(QMA6100P_REG_POWER_MODE, QMA6100P_MODE_NORMAL);
    vTaskDelay(pdMS_TO_TICKS(10));

    // Установка диапазона и частоты
    writeReg(QMA6100P_REG_RANGE, _range);
    writeReg(QMA6100P_REG_BW, QMA6100P_BW_250HZ);
    vTaskDelay(pdMS_TO_TICKS(10));

    ESP_LOGI(TAG, "QMA6100P initialized");
    return ESP_OK;
}

esp_err_t QMA6100P::configureAnyMotion(uint8_t threshold, uint8_t duration) {
    // threshold – значение для порога (зависит от диапазона)
    // duration – количество последовательных сэмплов для подтверждения движения (0..3)
    // Регистр Motion_Config (0x09): биты [7:6] – режим (any-motion), [5:4] – duration (0..3)
    uint8_t motion_cfg = (0x01 << 6) | ((duration & 0x03) << 4); // any-motion mode
    writeReg(QMA6100P_REG_MOTION_CFG, motion_cfg);

    // Регистр Motion_Threshold (0x0A)
    writeReg(QMA6100P_REG_MOTION_TH, threshold);

    // Включение прерывания any-motion в INT_EN_1 (0x16)
    uint8_t int_en1;
    readReg(QMA6100P_REG_INT_EN_1, &int_en1);
    int_en1 |= QMA6100P_INT_ANY_MOTION_EN;
    writeReg(QMA6100P_REG_INT_EN_1, int_en1);

    // Маппинг прерывания на пин INT1 через INT_MAP_1 (0x19)
    uint8_t int_map1;
    readReg(QMA6100P_REG_INT_MAP_1, &int_map1);
    int_map1 |= QMA6100P_INT1_ANY_MOTION;
    writeReg(QMA6100P_REG_INT_MAP_1, int_map1);

    ESP_LOGI(TAG, "Any-motion configured: threshold=%d, duration=%d", threshold, duration);
    return ESP_OK;
}

esp_err_t QMA6100P::readAccel(float &x, float &y, float &z) {
    uint8_t data[6];
    if (readRegs(QMA6100P_REG_ACCEL_X_L, data, 6) != ESP_OK) {
        return ESP_FAIL;
    }
    int16_t raw_x = (int16_t)(data[1] << 8 | data[0]);
    int16_t raw_y = (int16_t)(data[3] << 8 | data[2]);
    int16_t raw_z = (int16_t)(data[5] << 8 | data[4]);

    // Преобразование в g в зависимости от диапазона
    float scale;
    switch (_range) {
        case QMA6100P_RANGE_2G: scale = 16384.0f; break;
        case QMA6100P_RANGE_4G: scale = 8192.0f; break;
        case QMA6100P_RANGE_8G: scale = 4096.0f; break;
        case QMA6100P_RANGE_16G: scale = 2048.0f; break;
        default: scale = 16384.0f;
    }
    x = raw_x / scale;
    y = raw_y / scale;
    z = raw_z / scale;
    return ESP_OK;
}

bool QMA6100P::isMotionTriggered() {
    // Можно прочитать статус прерывания по регистру INT_STATUS_0 (0x0B)
    uint8_t status = 0;
    readReg(QMA6100P_REG_INT_STATUS_0, &status);
    return (status & QMA6100P_INT_ANY_MOTION_EN) != 0;
}

void AlzV5::initAccelerometer() {
    // I2C уже инициализирован в предыдущей части
    // (предположим, что i2c_driver_install выполнен ранее)
    accel = new QMA6100P(I2C_NUM_0, PIN_QMA_INT1);
    if (accel->init() != ESP_OK) {
        ESP_LOGE("AlzV5", "Failed to init QMA6100P");
        return;
    }
    // Настроим any-motion с умеренной чувствительностью
    accel->configureAnyMotion(20, 1); // threshold=20 (~0.3g при ±2g)
}

void AlzV5::enableWakeOnMotion() {
    if (!accel) return;

    // Настраиваем пин INT1 как вход с прерыванием по любому изменению (или по спаду/фронту)
    gpio_config_t io_conf = {};
    io_conf.pin_bit_mask = (1ULL << PIN_QMA_INT1);
    io_conf.mode = GPIO_MODE_INPUT;
    io_conf.pull_up_en = GPIO_PULLUP_ENABLE;   // на схеме internal pull-up, но можно и внешний
    io_conf.intr_type = GPIO_INTR_POSEDGE;     // прерывание по переднему фронту (активный высокий)
    gpio_config(&io_conf);

    // Установка обработчика прерывания (необязательно, можно только для wakeup)
    gpio_install_isr_service(0);
    gpio_isr_handler_add(PIN_QMA_INT1, motion_isr_handler, NULL);

    // Разрешить пробуждение из глубокого сна по этому пину
    esp_sleep_enable_gpio_wakeup();
    gpio_wakeup_enable(PIN_QMA_INT1, GPIO_INTR_HIGH_LEVEL); // или GPIO_INTR_POSEDGE

    ESP_LOGI("AlzV5", "Wake-on-motion enabled");
}

Встряхните плату – она должна проснуться и начать выполнение заново (или продолжить с того места, где был вызван сон, в зависимости от логики).

5. Возможные проблемы и их решение
I2C не работает – проверьте подтягивающие резисторы (на схеме 4.7 кОм), номера пинов, частоту I2C (можно уменьшить до 100 кГц для отладки).

Прерывание не срабатывает – убедитесь, что пин INT1 настроен правильно и акселерометр генерирует сигнал (можно прочитать регистр статуса вручную).

Пробуждение происходит сразу после сна – возможно, прерывание висит постоянно. Проверьте полярность: может быть, нужно настроить на низкий уровень или спадающий фронт. В даташите QMA6100P прерывание активно высокое (по умолчанию). Если после срабатывания оно остаётся высоким, нужно сбрасывать его чтением регистра статуса перед сном.

Выход из глубокого сна приводит к полному сбросу – при использовании esp_deep_sleep_start() система стартует заново, поэтому нужно иметь возможность быстро определить, что пробуждение было от акселерометра, и сразу перейти к обработке голоса, минуя долгую инициализацию. Это можно сделать, проверив причину пробуждения и загрузив конфигурацию из RTC-памяти.

Также необходимо учесть, что INT1 используется для any-motion (пробуждение по движению), а INT2 - для tap (пробуждение по удару). Но возможно, что в глубоком сне оба прерывания могут быть использованы для пробуждения, но ESP32-S3 может просыпаться только по одному источнику? На самом деле можно настроить пробуждение по нескольким GPIO одновременно (esp_sleep_enable_gpio_wakeup позволяет задать маску пинов с уровнем/фронтом). Так что можно разрешить пробуждение по обоим пинам.

Но тогда после пробуждения нужно определить, какой именно пин вызвал пробуждение. Для этого можно использовать esp_sleep_get_gpio_wakeup_status().

Далее, после пробуждения, если это INT2 (tap), то сразу инициируем вопрос "Что случилось?" и ожидаем голосовой ответ. При этом мы должны включить аудио, микрофон, установить соединение с сервером Xiaozhi и т.д.

Также нужно предусмотреть таймер на 1 минуту после пробуждения. Если за это время нет голосового ввода и нет any-motion (INT1 не срабатывал), то вызываем emergency(). Для отслеживания any-motion после пробуждения мы можем использовать прерывание INT1, но уже в нормальном режиме работы (не сна). Т.е. после пробуждения мы должны включить нормальный режим, обрабатывать прерывания INT1 как событие движения, которое сбрасывает таймер бездействия.

Таким образом, после пробуждения мы:

Определяем причину.

Если INT2 (tap) -> запускаем голосовой диалог (например, через API Xiaozhi) и одновременно запускаем таймер на 1 минуту.

Если INT1 (any-motion) -> возможно, просто просыпаемся и ждем голосовой команды? По условию "при отсутствии голосового контакта и активных движений в течении минуты (нет int1) перейти к emergency()". Это значит, что после пробуждения мы ожидаем либо голосовой ввод, либо движение (INT1). Если ни того, ни другого за минуту нет - emergency.

Значит, после пробуждения (любого) мы запускаем таймер на 60 секунд. При получении голосовой команды (через распознавание речи) или при срабатывании INT1 (движение) мы сбрасываем таймер и остаемся в активном режиме. Если таймер истекает - emergency.

Emergency() может быть функцией, которая отправляет сигнал (например, по Wi-Fi) на заданный сервер или зажигает светодиод, или воспроизводит звуковой сигнал.

Теперь детали по настройке QMA6100P на tap.

По даташиту QMA6100P:

Регистр 0x09: Motion_Config (биты [7:6] = 00 - нет, 01 - any-motion, 10 - tap, 11 - double-tap)

Регистр 0x0A: Motion_Threshold (порог для any-motion и tap)

Регистр 0x0B: Motion_Duration (для any-motion) и также используется для tap (tap duration?).

Регистр 0x0C: Tap_Threshold? Возможно отдельный регистр. Надо смотреть даташит. К сожалению, в предоставленном фрагменте нет полного даташита, но обычно для tap есть отдельные регистры: Tap_Threshold, Tap_Duration, Tap_Latency, Tap_Window.

Предположим, что для QMA6100P (похож на QMA7981) регистры tap:

0x1B: Tap_Threshold (по умолч. 0x10)

0x1C: Tap_Duration (макс. время удара)

0x1D: Tap_Latency (время между ударами для double-tap)

0x1E: Tap_Window (окно для double-tap)

Надо найти точную документацию. Но для целей ответа мы можем предложить общий подход с комментариями, что конкретные значения регистров нужно уточнить по даташиту.

В коде драйвера добавим функцию configureTap():

cpp
esp_err_t QMA6100P::configureTap(uint8_t threshold, uint8_t duration, bool double_tap = false) {
    // Регистр Motion_Config (0x09) устанавливаем режим tap (10) или double-tap (11)
    uint8_t motion_cfg = (double_tap ? 0x03 : 0x02) << 6; // биты 7-6
    writeReg(QMA6100P_REG_MOTION_CFG, motion_cfg);

    // Порог tap (регистр 0x1B)
    writeReg(0x1B, threshold);
    // Длительность tap (регистр 0x1C)
    writeReg(0x1C, duration);
    // Возможно, также нужно настроить латентность и окно для double-tap

    // Включение прерывания tap в INT_EN_1 или INT_EN_2?
    // Скорее всего, биты включения tap находятся в INT_EN_2 (0x17)
    uint8_t int_en2;
    readReg(0x17, &int_en2);
    int_en2 |= (1 << 0); // предположим, бит 0 для tap
    writeReg(0x17, int_en2);

    // Маппинг на INT2: регистр INT_MAP_2 (0x1A)
    uint8_t int_map2;
    readReg(0x1A, &int_map2);
    int_map2 |= (1 << 0); // маппим tap на INT2
    writeReg(0x1A, int_map2);
    return ESP_OK;
}

1. Доработка драйвера QMA6100P для поддержки tap‑детекции
В файл qma6100p.h добавим определения регистров для tap (уточните по даташиту вашей версии QMA6100P; здесь используются типовые адреса, аналогичные QMA7981):

// Регистры для настройки tap (предположительно)
#define QMA6100P_REG_TAP_THRESHOLD   0x1B
#define QMA6100P_REG_TAP_DURATION    0x1C
#define QMA6100P_REG_TAP_LATENCY     0x1D   // для double-tap
#define QMA6100P_REG_TAP_WINDOW      0x1E   // для double-tap
#define QMA6100P_REG_INT_EN_2        0x17   // второй регистр разрешения прерываний
#define QMA6100P_INT_TAP_EN           (1 << 0)   // бит включения tap (предположительно)

esp_err_t QMA6100P::configureTap(uint8_t threshold, uint8_t duration, bool double_tap) {
    // Установка режима tap/double-tap в Motion_Config (0x09)
    uint8_t motion_cfg;
    readReg(QMA6100P_REG_MOTION_CFG, &motion_cfg);
    motion_cfg &= 0x3F;               // очистить биты 7-6
    motion_cfg |= (double_tap ? 0x03 : 0x02) << 6; // 10 = tap, 11 = double-tap
    writeReg(QMA6100P_REG_MOTION_CFG, motion_cfg);

    // Порог и длительность
    writeReg(QMA6100P_REG_TAP_THRESHOLD, threshold);
    writeReg(QMA6100P_REG_TAP_DURATION, duration);

    // Для double-tap можно дополнительно настроить latency и window
    if (double_tap) {
        // Примерные значения (нужно подобрать)
        writeReg(QMA6100P_REG_TAP_LATENCY, 0x10); // задержка между ударами
        writeReg(QMA6100P_REG_TAP_WINDOW,  0x20); // окно для второго удара
    }

    // Включение прерывания tap в INT_EN_2
    uint8_t int_en2;
    readReg(QMA6100P_REG_INT_EN_2, &int_en2);
    int_en2 |= QMA6100P_INT_TAP_EN;
    writeReg(QMA6100P_REG_INT_EN_2, int_en2);

    // Маппинг tap на INT2 через INT_MAP_2 (0x1A)
    uint8_t int_map2;
    readReg(0x1A, &int_map2);
    int_map2 |= QMA6100P_INT_TAP_EN;   // тот же бит маппинга (предположительно)
    writeReg(0x1A, int_map2);

    return ESP_OK;
}
Расчёт порога 1.25g:
Если используется диапазон ±2g, то 1g соответствует 16384 LSB. 1.25g = 1.25 * 16384 = 20480 LSB. Но регистр порога обычно 8-битный и задаётся в единицах, где 1 LSB = ? Зависит от диапазона. В даташите QMA6100P для tap‑порога, скорее всего, значение задаётся в мг/LSB. Например, при диапазоне ±2g разрешение может быть 4 мг/LSB, тогда 1.25g = 1250 мг / 4 = 312.5 → округлим до 0x138 (312 десятичное). Но обычно регистр 8-битный, значит придётся выбрать другое разрешение или использовать диапазон ±16g с меньшим разрешением. Точные значения нужно брать из документации. Для примера установим threshold = 40 (что примерно 0.16g при 4 мг/LSB, но это слишком мало). Лучше ориентироваться на реальный даташит. В коде ниже мы просто используем переменную tap_threshold, которую пользователь должен вычислить.

3. Настройка пробуждения из глубокого сна по INT1 и INT2
В функции enableWakeOnMotion() (лучше переименовать в enableWakeOnInterrupts()) настроим оба пина для пробуждения:

cpp
void AlzV5::enableWakeOnInterrupts() {
    if (!accel) return;

    // Настраиваем оба пина как входы с подтяжкой (согласно схеме)
    gpio_config_t io_conf = {};
    io_conf.pin_bit_mask = (1ULL << PIN_QMA_INT1) | (1ULL << PIN_QMA_INT2);
    io_conf.mode = GPIO_MODE_INPUT;
    io_conf.pull_up_en = GPIO_PULLUP_ENABLE;   // на INT1 internal pull-up, на INT2 вероятно тоже
    io_conf.intr_type = GPIO_INTR_POSEDGE;     // активный высокий уровень (уточнить по даташиту)
    gpio_config(&io_conf);

    // Устанавливаем оба пина как источники пробуждения
    esp_sleep_enable_gpio_wakeup();
    gpio_wakeup_enable(PIN_QMA_INT1, GPIO_INTR_HIGH_LEVEL);  // или GPIO_INTR_POSEDGE
    gpio_wakeup_enable(PIN_QMA_INT2, GPIO_INTR_HIGH_LEVEL);

    ESP_LOGI("AlzV5", "Wake-up by INT1 (any-motion) and INT2 (tap) enabled");
}

    // Инициализировать I2S для микрофона и динамика (если не было)
    // Воспроизвести вопрос "Что случилось?" (через API Xiaozhi или локальный файл)
    play_sound_q();   // функция, которую нужно реализовать

    // Сбросить флаг движения
    motion_detected = false;

    // Создать таймер на 60 секунд, если ещё не создан
    if (inactivity_timer == NULL) {
        inactivity_timer = xTimerCreate("inactivity", pdMS_TO_TICKS(60000), pdFALSE, NULL, inactivity_timer_callback);
    }
    xTimerReset(inactivity_timer, 0);  // перезапустить таймер

    // Настроить прерывание INT1 для сброса таймера при движении
    gpio_isr_handler_add(PIN_QMA_INT1, motion_isr, NULL);
    // Также нужно обрабатывать прерывание в задаче, чтобы сбросить таймер
    // Можно создать задачу, которая ждет флаг motion_detected
    xTaskCreate(motion_task, "motion_task", 2048, NULL, 5, NULL);
}

static void motion_task(void* arg) {
    while (1) {
        if (motion_detected) {
            motion_detected = false;
            // Сброс таймера бездействия
            if (inactivity_timer) {
                xTimerReset(inactivity_timer, 0);
            }
            ESP_LOGI("MOTION", "Motion detected, inactivity timer reset");
        }
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}
Где play_sound_q() – функция, которая ставит в очередь воспроизведения аудио. В реальном проекте Xiaozhi, вероятно, используется API для отправки текста на сервер и получения аудио. Можно просто отправить команду "скажи 'Что случилось?'" через основной цикл обработки.

Также нужно обрабатывать голосовой ввод: когда пользователь что-то сказал, распознавание должно сбросить таймер. Для этого в коде обработки голосовой команды (где-то в main) добавим вызов reset_inactivity_timer(). Эту функцию можно реализовать аналогично сбросу таймера.

cpp
void reset_inactivity_timer() {
    if (inactivity_timer) {
        xTimerReset(inactivity_timer, 0);
    }
}
Функция emergency() может быть определена так:

cpp
void emergency() {
    ESP_LOGI("EMERGENCY", "No response for 1 minute, entering emergency mode");
    // Например, отправить сигнал тревоги по Wi-Fi, зажечь LED и т.п.
    gpio_set_level(PIN_LED_ON, 1);  // включить индикатор
    // Воспроизвести звуковой сигнал
    play_sound("emergency");

6. Интеграция с Xiaozhi
В исходном коде Xiaozhi, скорее всего, есть основной цикл, который обрабатывает голосовые команды. Нужно добавить вызов reset_inactivity_timer() при успешном распознавании речи или при получении команды от пользователя. Также нужно убедиться, что после пробуждения по INT2 мы переходим в режим ожидания голоса, а не уходим сразу в сон.

Возможно, придется модифицировать конечный автомат состояний проекта.

7. Проверка порогов
Для точной настройки порога 1.25g необходимо изучить даташит QMA6100P и вычислить значение регистра. Обычно порог задаётся в милли-g и имеет разрешение, например, 62.5 мг/LSB. Тогда 1.25g = 1250 мг / 62.5 = 20. То есть значение 0x14. Если разрешение другое, подбираем. Также длительность удара задаётся в количестве выборок или миллисекундах. Экспериментируйте.

8. Полный цикл работы
Плата в глубоком сне.

Происходит удар с ускорением >1.25g → INT2 активен → ESP32 просыпается.

В app_main определяем причину: INT2.

Вызываем handleTapWake():

Включаем аудио.

Воспроизводим "Что случилось?".

Запускаем таймер 60 секунд.

Включаем обработку движений (INT1) и голоса.

Если в течение 60 секунд:

Есть движение (INT1) → сбрасываем таймер.

Есть голосовая команда → сбрасываем таймер и обрабатываем команду.

Если таймер истек → emergency().

Если после emergency прошло время без действий, можно снова уснуть.
